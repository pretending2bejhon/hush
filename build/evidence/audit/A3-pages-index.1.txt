// The Handoff, as section 18 spells it.
const spec = fs.readFileSync('build/HUSH-SPEC.md', 'utf8');
const handoff = spec.slice(spec.indexOf('## 18. Handoff'));
const want = {
  title: /the page title: `([^`]+)`/.exec(handoff)[1],
  link: /the link: `([^`]+)`, to `([^`]+)`/.exec(handoff).slice(1, 3),
  notice: /the notice: `([^`]+)`/.exec(handoff)[1],
  calibration: /while the calibration is up, `([^`]+)` in `<span id="calibration">`/.exec(handoff)[1],
  accent: /in the accent (#[0-9a-fA-F]{6})/.exec(handoff)[1],
};
console.log('Handoff wants:', JSON.stringify(want));

function judge(label, html) {
  const title = (html.match(/<title>([^<]*)<\/title>/) || [])[1];
  const body = (html.match(/<body>([\s\S]*)<\/body>/) || [])[1] || '';
  const footer = (body.match(/<footer>([\s\S]*?)<\/footer>/) || [])[1] || '';
  const items = footer.split('\n').map((l) => l.trim()).filter(Boolean);
  const outside = body.replace(/<footer>[\s\S]*?<\/footer>/, '').replace(/<script[\s\S]*?<\/script>/g, '').trim();
  const expect = [
    `<span id="calibration">${want.calibration}</span>`,
    `<a href="${want.link[1]}">${want.link[0]}</a>`,
    `<span>${want.notice}</span>`,
    '<output id="playtest" hidden></output>',
  ];
  const accent = new RegExp(`#calibration\\s*\\{\\s*color:\\s*${want.accent}\\s*;?\\s*\\}`, 'i').test(html);
  const scripts = html.match(/<script[^>]*>/g) || [];
  const externals = [...html.matchAll(/(?:src|href)="([^"]+)"/g)].map((m) => m[1]).filter((u) => /^(https?:)?\/\//.test(u));
  console.log(`${label}: title ${JSON.stringify(title)} equal: ${title === want.title}`);
  console.log(`${label}: footer items exactly the Handoff's, in order: ${JSON.stringify(items) === JSON.stringify(expect)}${JSON.stringify(items) === JSON.stringify(expect) ? '' : ` (found ${JSON.stringify(items)})`}`);
  console.log(`${label}: #calibration in the accent ${want.accent}: ${accent} | the body beside the footer: ${JSON.stringify(outside)}`);
  console.log(`${label}: script tags ${JSON.stringify(scripts)} | absolute URLs in src or href: ${JSON.stringify(externals)} | hush-build meta: ${(html.match(/<meta name="hush-build" content="([0-9a-f]+)"/) || [])[1] || 'none'}`);
}
