// Audit 3 scratch check, run from the repo root: exact pins, the committed lockfile, its sources and integrity,
// and every install script, from the lockfile's flags and from the installed package.json files.
import { execFileSync, spawnSync } from 'node:child_process';
import fs from 'node:fs';
import path from 'node:path';

const git = (a) => execFileSync('git', a, { encoding: 'utf8' }).trim();
const lock = JSON.parse(fs.readFileSync('package-lock.json', 'utf8'));
const pkg = JSON.parse(fs.readFileSync('package.json', 'utf8'));
console.log('package-lock.json tracked:', git(['ls-files', '--', 'package-lock.json']) === 'package-lock.json');
console.log('package-lock.json and package.json equal HEAD:', spawnSync('git', ['diff', '--quiet', 'HEAD', '--', 'package-lock.json', 'package.json']).status === 0);
console.log('.npmrc tracked:', git(['ls-files', '--', '.npmrc']) === '.npmrc', '| settings:', fs.readFileSync('.npmrc', 'utf8').trim().split(/\r?\n/).join(', '));
console.log('lockfileVersion:', lock.lockfileVersion, '| lock name and version equal package.json:', lock.name === pkg.name && lock.version === pkg.version);
const exact = /^[0-9]+\.[0-9]+\.[0-9]+$/;
let loose = 0;
let drift = 0;
const root = lock.packages[''];
for (const k of ['dependencies', 'devDependencies']) {
  for (const [name, want] of Object.entries(pkg[k] || {})) {
    if (!exact.test(want)) { loose++; console.log('NOT EXACT', k, name, want); }
    const r = (root[k] || {})[name];
    const inst = (lock.packages[`node_modules/${name}`] || {}).version;
    if (r !== want || inst !== want) { drift++; console.log('DRIFT', name, want, r, inst); }
    console.log(`  ${k === 'dependencies' ? 'dep   ' : 'devdep'} ${name}@${want} lock root ${r} locked ${inst}`);
  }
}
for (const k of ['optionalDependencies', 'peerDependencies', 'bundleDependencies', 'bundledDependencies', 'overrides', 'resolutions']) {
  if (pkg[k]) console.log('package.json also has', k, JSON.stringify(pkg[k]));
}
console.log('engines:', JSON.stringify(pkg.engines), '| .nvmrc:', fs.readFileSync('.nvmrc', 'utf8').trim(), '| node:', process.version);
console.log('package.json ranges that are not exact versions:', loose);
console.log('direct deps whose lock root or locked version differs from package.json:', drift);
let n = 0;
let prod = 0;
let noInt = 0;
let offReg = 0;
const flagged = [];
const scripted = [];
for (const [key, p] of Object.entries(lock.packages)) {
  if (!key) continue;
  n++;
  if (p.link) { console.log('LINK', key); continue; }
  if (!p.dev) prod++;
  if (!p.integrity || !/^sha512-/.test(p.integrity)) { noInt++; console.log('NO SHA-512 INTEGRITY', key, p.integrity); }
  if (!/^https:\/\/registry\.npmjs\.org\//.test(p.resolved || '')) { offReg++; console.log('NOT REGISTRY', key, p.resolved); }
  const tags = `${p.dev ? ' dev' : ' prod'}${p.optional || p.devOptional ? ' optional' : ''}${p.os ? ` os=${p.os.join('/')}` : ''}`;
  if (p.hasInstallScript) flagged.push(`${key}${tags}`);
  const manifest = path.join(key, 'package.json');
  if (fs.existsSync(manifest)) {
    const s = JSON.parse(fs.readFileSync(manifest, 'utf8')).scripts || {};
    const life = ['preinstall', 'install', 'postinstall', 'prepare'].filter((x) => s[x]);
    if (life.length) scripted.push(`${key} (${life.join(', ')})${tags}`);
  }
}
console.log('locked packages:', n, '| production:', prod, '| without sha512 integrity:', noInt, '| resolved off registry.npmjs.org:', offReg);
