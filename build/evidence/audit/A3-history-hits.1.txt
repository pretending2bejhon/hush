// Audit 3 scratch check, run from the repo root: the history scan's hits explained without printing what they matched.
// For each blob given, every email address in it is hashed (sha256 of its lowercase form) and looked up in
// tools/public-patterns.mjs's THIRD_PARTY_EMAIL_HASHES; every token-shaped hit is shown as its line number and whether
// that line is a regular-expression or grep-pattern source (a literal prefix followed by a character class).
import { execFileSync } from 'node:child_process';
import crypto from 'node:crypto';
import fs from 'node:fs';

const pp = fs.readFileSync('tools/public-patterns.mjs', 'utf8');
const block = /THIRD_PARTY_EMAIL_HASHES = new Set\(\[([\s\S]*?)\]\)/.exec(pp)[1];
const allowed = new Set(block.match(/[0-9a-f]{64}/g));
const EMAIL = /[A-Za-z0-9._%+-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)*\.[A-Za-z]{2,}/g;
const TOKEN = /ghp_[A-Za-z0-9]{20,}|github_pat_|gh[ousr]_[A-Za-z0-9]{20,}|npm_[A-Za-z0-9]{30,}|AKIA[0-9A-Z]{16}|-----BEGIN [A-Z ]*PRIVATE KEY|xox[abprs]-[A-Za-z0-9-]{10,}|sk-[A-Za-z0-9_-]{20,}|AIza[0-9A-Za-z_-]{35}|_authToken/g;
for (const sha of process.argv.slice(2)) {
  const text = execFileSync('git', ['cat-file', '-p', sha], { encoding: 'utf8', maxBuffer: 1e9 });
  const lines = text.split('\n');
  for (const m of text.match(EMAIL) || []) {
    const h = crypto.createHash('sha256').update(m.toLowerCase()).digest('hex');
    console.log(`blob ${sha}: an email, sha256 ${h.slice(0, 16)}..., in THIRD_PARTY_EMAIL_HASHES: ${allowed.has(h)}`);

    }
  });
}
